```go
func (s *TerminalService) Write(sessionId string, data string) error {
    ss, err := s.resolveSession(sessionId)
    if err != nil { return err }
    ss.mu.Lock(); defer ss.mu.Unlock()
    if !ss.running {
        if err := s.startSessionLocked(ss, int(ss.lastSize.Cols), int(ss.lastSize.Rows)); err != nil {
            return err
        }
    }
    if ss.ptmx == nil { return fmt.Errorf("terminal not started") }
    b := []byte(data)
    for len(b) > 0 {
        n, err := ss.ptmx.Write(b); if err != nil { return err }
        b = b[n:]
    }
    return nil
}
```
Do NOT call `Start()` separately before `Write()`. Auto-resume is built in.

func (s *TerminalService) GetActiveSession() *SessionInfo {
    s.mu.RLock(); defer s.mu.RUnlock()
    if s.activeSessionID == "" { return nil }
    ss, ok := s.sessions[s.activeSessionID]

}

| T-24-01 | Denial of Service | TerminalService.Write called with malicious long input | accept | The cmdLine is built from user-saved script content (trusted source — user saved it via the editor).
